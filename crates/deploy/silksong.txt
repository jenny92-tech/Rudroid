# Silksong Title Deploy Selection profile (ticket 33).
#
# One capability id per line; `#` starts a comment. Ids match the C++ plugin
# ids and equal the deployed file names (see crates/docs/MIGRATION-MAP.md; the
# reply_catalog artifact is resolved via the explicit deploy-select alias).
# Use with:
#
#   bash crates/scripts/docker-rust.sh build --release --sdl
#   python3 crates/scripts/deploy-select.py \
#       --build crates/target/docker-linux-arm64/target/release \
#       --dest /tmp/kit-silksong --from-profile crates/deploy/silksong.txt \
#       --include-loader
#   python3 crates/scripts/kit-manifest.py scan \
#       --build /tmp/kit-silksong --output /tmp/silksong-kit.json
# Then copy that complete selected kit and manifest to the ARM64 device;
# run its loader --check there (the guest binary does not run on macOS).
#
# Redeploying over an existing dir removes this script's stale managed
# members and never touches user data (ticket 58 semantics).
# The device Host Config is used AS-IS (schema compatibility is a hard rule;
# the TOML gains no new required keys). The profile decides what gets copied;
# without a [plugins] allowlist the loader loads everything it finds.

platform_log_stderr

android_base

# provider every RegisterJni phase resolves through. Without it android_base
# misses getStreamVolume and fails the phase (ticket 76 red chain A).
reply_catalog

android_elf_load

android_bionic

platform_android_asset_packs

unity_6

il2cpp_patch

platform_sdl_runtime

sdk_fmod

silksong_unity6
